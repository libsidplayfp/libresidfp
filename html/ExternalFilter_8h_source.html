136int32_t ExternalFilter::clock(int32_t input)
137{
138 const int32_t Vi = input << 11;
139 const int32_t dVlp = (w0lp_1_s7 * (Vi - Vlp) >> 7);
140 const int32_t dVhp = (w0hp_1_s17 * (Vlp - Vhp) >> 17);
141 Vlp += dVlp;
142 Vhp += dVhp;
143 return (Vlp - Vhp) >> 11;
144}
