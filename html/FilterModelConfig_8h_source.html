147 inline void buildSummerTable(const OpAmp& opampModel)
148 {
149 const double r_N16 = 1. / N16;
150
151 int idx = 0;
152 for (int i = 0; i < 5; i++)
153 {
154 const int idiv = 2 + i; // 2 - 6 input "resistors".
155 const int size = idiv << 16;
156 const double n = idiv;
157 const double r_idiv = 1. / idiv;
158 opampModel.reset();
159
160 for (int vi = 0; vi < size; vi++)
161 {
162 const double vin = vmin + vi * r_N16 * r_idiv; /* vmin .. vmax */
163 summer[idx++] = getNormalizedValue(opampModel.solve(n, vin));
164 }
165 }
166 }

176 inline void buildMixerTable(const OpAmp& opampModel, double nRatio)
177 {
178 const double r_N16 = 1. / N16;
179
180 int idx = 0;
181 for (int i = 0; i < 8; i++)
182 {
183 const int idiv = (i == 0) ? 1 : i;
184 const int size = (i == 0) ? 1 : i << 16;
185 const double n = i * nRatio;
186 const double r_idiv = 1. / idiv;
187 opampModel.reset();
188
189 for (int vi = 0; vi < size; vi++)
190 {
191 const double vin = vmin + vi * r_N16 * r_idiv; /* vmin .. vmax */
192 mixer[idx++] = getNormalizedValue(opampModel.solve(n, vin));
193 }
194 }
195 }

204 inline void buildVolumeTable(const OpAmp& opampModel, double nDivisor)
205 {
206 const double r_N16 = 1. / N16;
207
208 int idx = 0;
209 for (int n8 = 0; n8 < 16; n8++)
210 {
211 constexpr int size = 1 << 16;
212 const double n = n8 / nDivisor;
213 opampModel.reset();
214
215 for (int vi = 0; vi < size; vi++)
216 {
217 const double vin = vmin + vi * r_N16; /* vmin .. vmax */
218 volume[idx++] = getNormalizedValue(opampModel.solve(n, vin));
219 }
220 }
221 }
222
230 inline void buildResonanceTable(const OpAmp& opampModel, const double resonance_n[16])
231 {
232 const double r_N16 = 1. / N16;
233
234 int idx = 0;
235 for (int n8 = 0; n8 < 16; n8++)
236 {
237 constexpr int size = 1 << 16;
238 opampModel.reset();
239
240 for (int vi = 0; vi < size; vi++)
241 {
242 const double vin = vmin + vi * r_N16; /* vmin .. vmax */
243 resonance[idx++] = getNormalizedValue(opampModel.solve(resonance_n[n8], vin));
244 }
245 }
246 }
