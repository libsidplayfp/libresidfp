219RESIDFP_INLINE
220uint16_t Filter::clock(Voice& voice1, Voice& voice2, Voice& voice3)
221{
222 const int32_t V1 = getNormalizedVoice(voice1);
223 const int32_t V2 = getNormalizedVoice(voice2);
224 // Voice 3 is silenced by voice3off if it is not routed through the filter.
225 const int32_t V3 = (filt3 || !voice3off) ? getNormalizedVoice(voice3) : getSilentVoice(voice3);
226
227 int32_t Vsum = 0;
228 int32_t Vmix = 0;
229
230 (filt1 ? Vsum : Vmix) += V1;
231 (filt2 ? Vsum : Vmix) += V2;
232 (filt3 ? Vsum : Vmix) += V3;
233 (filtE ? Vsum : Vmix) += Ve;
234
235 Vhp = currentSummer[currentResonance[Vbp] + Vlp + Vsum];
236
237 Vmix += solveIntegrators();
238
239 return currentVolume[currentMixer[Vmix]];
240}
