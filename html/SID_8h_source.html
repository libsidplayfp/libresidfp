351RESIDFP_INLINE
352int SID::clock(unsigned int cycles, short* buf)
353{
354 assert(buf);
355
356 ageBusValue(cycles);
357 int s = 0;
358
359 while (cycles != 0)
360 {
361 unsigned int delta_t = std::min(nextVoiceSync, cycles);
362
363 if (likely(delta_t > 0))
364 {
365 for (unsigned int i = 0; i < delta_t; i++)
366 {
367 clockWaveGen();
368 clockEnvGen();
369
370 int output = clockFilt();
371 if (unlikely(resampler->input(output)))
372 {
373 buf[s++] = resampler->getOutput(scaleFactor);
374 }
375 }
376
377 cycles -= delta_t;
378 nextVoiceSync -= delta_t;
379 }
380
381 if (unlikely(nextVoiceSync == 0))
382 {
383 voiceSync(true);
384 }
385 }
386
387 return s;
388}

390RESIDFP_INLINE
391int SID::clock(short* buf, int bufSize)
392{
393 assert(buf);
394 assert(bufSize > 0);
395
396 int cycles = 0;
397
398 for (int s = 0; s < bufSize;)
399 {
400 unsigned int delta_t = nextVoiceSync;
401
402 if (likely(delta_t > 0))
403 {
404 unsigned int i = 0;
405 do
406 {
407 i++;
408 clockWaveGen();
409 clockEnvGen();
410
411 int output = clockFilt();
412 if (unlikely(resampler->input(output)))
413 {
414 buf[s++] = resampler->getOutput(scaleFactor);
415 if (unlikely(s == bufSize))
416 {
417 break;
418 }
419 }
420 }
421 while (i < delta_t);
422
423 cycles += i;
424 nextVoiceSync -= i;
425 }
426
427 if (likely(nextVoiceSync == 0))
428 {
429 voiceSync(true);
430 }
431 }
432
433 ageBusValue(cycles);
434
435 return cycles;
436}
